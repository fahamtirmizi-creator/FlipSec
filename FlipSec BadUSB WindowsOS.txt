
REM FLIPSEC - WINDOWS SECURITY SCANNER
REM Windows 10 / 11 - Flipper Zero BadUSB
REM Requires an unlocked Windows computer

DELAY 2500

REM STEP 1 - OPEN WINDOWS RUN
GUI r
DELAY 1800

REM STEP 2 - CLEAR RUN FIELD
CTRL a
DELAY 400

REM STEP 3 - LAUNCH POWERSHELL
STRING powershell.exe
DELAY 1500
ENTER

REM STEP 4 - WAIT FOR POWERSHELL
DELAY 8000

REM STEP 5 - CREATE SECURITY REPORT
STRING $d=[Environment]::GetFolderPath('Desktop'); $f=Join-Path $d ('flipsec_'+(Get-Date -Format 'HH-mm-ss_dd-MM-yyyy')+'.txt'); '=== FLIPSEC WINDOWS SECURITY REPORT ===' | Out-File $f; Get-Date | Out-File $f -Append; '=== SYSTEM ===' | Out-File $f -Append; Get-CimInstance Win32_OperatingSystem | Select-Object Caption,Version,BuildNumber | Out-File $f -Append; '=== USER ===' | Out-File $f -Append; whoami | Out-File $f -Append; '=== PROCESSES ===' | Out-File $f -Append; Get-Process | Select-Object ProcessName,Id,CPU | Out-File $f -Append; '=== LISTENING PORTS ===' | Out-File $f -Append; Get-NetTCPConnection -State Listen | Out-File $f -Append; '=== NETWORK ===' | Out-File $f -Append; Get-NetTCPConnection | Out-File $f -Append; '=== FIREWALL ===' | Out-File $f -Append; Get-NetFirewallProfile | Select-Object Name,Enabled | Out-File $f -Append; '=== DEFENDER ===' | Out-File $f -Append; Get-MpComputerStatus | Select-Object AntivirusEnabled,RealTimeProtectionEnabled | Out-File $f -Append; '=== BITLOCKER ===' | Out-File $f -Append; Get-BitLockerVolume | Select-Object MountPoint,ProtectionStatus | Out-File $f -Append; '=== SCAN COMPLETE ===' | Out-File $f -Append; Start-Process notepad.exe -ArgumentList ('"'+$f+'"')
ENTER
